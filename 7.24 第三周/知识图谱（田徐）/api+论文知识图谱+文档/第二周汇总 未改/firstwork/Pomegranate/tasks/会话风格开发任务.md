| 放在 Rust 侧 | 放在 React 侧 |
|-------------|--------------|

```rust
use serde::{Deserialize, Serialize};

/// 任务执行会话
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct TaskSession {
    pub id: String,            // UUID
    pub plan_path: String,     // 计划文件路径
    pub plan_name: String,     // 计划名称（从文件名/标题提取）
    pub status: SessionStatus, // idle | running | waiting_confirm | paused | completed
    pub current_phase_index: i32,
    pub total_phases: i32,
    pub created_at: String,
    pub updated_at: String,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub enum SessionStatus {
    #[serde(rename = "idle")]
    Idle,
    #[serde(rename = "running")]
    Running,
    #[serde(rename = "waiting_confirm")]
    WaitingConfirm,
    #[serde(rename = "paused")]
    Paused,
    #[serde(rename = "completed")]
    Completed,
}

/// 执行阶段
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ExecutionPhase {
    pub id: String,            // "phase_0"
    pub session_id: String,
    pub index_num: i32,
    pub name: String,          // "环境准备"
    pub description: String,   // 阶段描述
    pub status: PhaseStatus,
    pub files_modified: Option<String>, // JSON 数组字符串
    pub result_summary: Option<String>,
    pub started_at: Option<String>,
    pub finished_at: Option<String>,
}

#[derive(Debug, Clone, Serialize, Deserialize, PartialEq)]
pub enum PhaseStatus {
    #[serde(rename = "pending")]
    Pending,
    #[serde(rename = "running")]
    Running,
    #[serde(rename = "waiting_confirm")]
    WaitingConfirm,
    #[serde(rename = "completed")]
    Completed,
    #[serde(rename = "skipped")]
    Skipped,
    #[serde(rename = "failed")]
    Failed,
}

/// 执行日志
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ExecutionLog {
    pub id: i64,
    pub session_id: String,
    pub phase_id: Option<String>,
    pub level: String,         // info | warn | error
    pub message: String,
    pub created_at: String,
}

/// 计划文件解析结果（不存库，仅用于传输）
#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct ParsedPlan {
    pub name: String,
    pub phases: Vec<PhaseDraft>,
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PhaseDraft {
    pub id: String,
    pub name: String,
    pub description: String,
}
```

### 二、后端：服务层

#### 任务 2.1：计划文件解析服务

**文件**：`src-tauri/src/services/session_plan.rs`（新建）

**功能**：
- `parse_plan_file(path: &str) -> Result<ParsedPlan, AppError>`
  - 读取 `.md` 文件
  - 用正则识别 Phase/步骤/任务标记：
    - `Phase N`、`Phase N：`、`Phase N:`
    - `步骤 N`、`Step N`
    - `任务 N`、`Task N`
    - `### Phase N: 名称`（Markdown 标题格式）
  - 提取阶段名称和描述
  - 返回 `ParsedPlan { name, phases: Vec<PhaseDraft> }`
- 支持文件格式：`.md`, `.txt`（纯文本逐行匹配）

**边界处理**：
- 文件不存在 → `AppError::NotFound`
- 无法解析到任何 Phase → 返回空 phases + warning
- 文件编码非 UTF-8 → 尝试自动检测编码

#### 任务 2.2：会话管理服务

**文件**：`src-tauri/src/services/session_manager.rs`（新建）

**功能**：
- `create_session(db, plan_path) -> Result<TaskSession>`
  - 调 `parse_plan_file()` 解析计划
  - 生成 UUID
  - 创建 session + phases 入库
  - 返回 session

- `start_phase(db, session_id, phase_index) -> Result<ExecutionPhase>`
  - 校验 session 状态
  - 更新当前 phase 为 running
  - 记录 started_at

- `complete_phase(db, session_id, phase_index, result_summary, files_modified) -> Result<()>`
  - 更新 phase 状态为 completed/waiting_confirm
  - 记录 finished_at, result_summary

- `confirm_phase(db, session_id) -> Result<()>`
  - 将当前 waiting_confirm 的 phase 标记为 completed
  - 推进到下一 phase

- `skip_phase(db, session_id, phase_index) -> Result<()>`
  - 标记 phase 为 skipped

- `retry_phase(db, session_id, phase_index) -> Result<()>`
  - 重置 phase 状态为 pending，清除之前的结果

- `pause_session(db, session_id) -> Result<()>`
- `resume_session(db, session_id) -> Result<()>`

**提示词模板**（存放在代码常量中）：

```
你正在执行一个多阶段开发任务。当前任务计划如下：

{plan_content}

执行规则：
1. 从 Phase {current_phase} 开始执行

5. 必须等待用户确认后才能继续下一 Phase
6. 如果遇到错误，立即停止并报告

当前状态：正在执行 Phase {current_phase_name}
```

**上下文构建**：
- `build_session_context(session, phases)` → 拼接当前计划 + 已完成 Phase 摘要 + 当前 Phase 信息
- 在 `send_ai_message` 前注入 system prompt
